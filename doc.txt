The Moat Academy guide explicitly calls for a user structure containing information such as:

```text

email
```

Create:

```text

```

```python
from datetime import datetime, timezone

from werkzeug.security import generate_password_hash, check_password_hash

from app.extensions import db


class User(db.Model):
    __tablename__ = "users"

    id = db.Column(db.Integer, primary_key=True)

    email = db.Column(
        db.String(120),
        unique=True,
        nullable=False,
        index=True
    )

    password_hash = db.Column(
        db.String(255),
        nullable=False
    )

    role = db.Column(
        db.String(20),
        nullable=False,
        default="patient"
    )

    created_at = db.Column(
        db.DateTime,
        default=lambda: datetime.now(timezone.utc),
        nullable=False
    )

    def set_password(self, password):
        self.password_hash = generate_password_hash(password)

    def check_password(self, password):
        return check_password_hash(
            self.password_hash,
            password
        )

    def __repr__(self):
        return f"<User {self.email}>"
```

Notice that we **never store the original password**.

That is specifically required by the teaching guide.

# 13. Step 11 — Implement registration

```text
app/auth/routes.py
```

```python
from flask import (
    render_template,
    redirect,
    url_for,
    flash,
    session
)

from app.auth import auth_bp
from app.auth.forms import RegistrationForm, LoginForm
from app.extensions import db
from app.models.user import User


@auth_bp.route("/register", methods=["GET", "POST"])
def register():

    if "user_id" in session:
        return redirect(url_for("main.index"))

    form = RegistrationForm()

    if form.validate_on_submit():

        existing_user = User.query.filter_by(
            email=form.email.data.lower().strip()
        ).first()

        if existing_user:
            flash(
                "An account with that email already exists.",
                "danger"
            )
            return render_template(
                "auth/register.html",
                form=form
            )

        user = User(
            email=form.email.data.lower().strip(),
            role="patient"
        )

        user.set_password(form.password.data)

        db.session.add(user)
        db.session.commit()

        flash(
            "Registration successful. Please log in.",
            "success"
        )

        return redirect(url_for("auth.login"))

    return render_template(
        "auth/register.html",
        form=form
    )
```

The guide says a newly registered user should receive appropriate feedback such as **“Registration successful. Please log in.”**

# 14. Step 12 — Implement login

Add to:

```text
app/auth/routes.py
```

```python
@auth_bp.route("/login", methods=["GET", "POST"])
def login():

    if "user_id" in session:
        return redirect(url_for("main.index"))

    form = LoginForm()

    if form.validate_on_submit():

        email = form.email.data.lower().strip()

        user = User.query.filter_by(
            email=email
        ).first()

        if user is None or not user.check_password(
            form.password.data
        ):
            flash(
                "Invalid email or password.",
                "danger"
            )

            return render_template(
                "auth/login.html",
                form=form
            )

        session.clear()

        session["user_id"] = user.id
        session["role"] = user.role

        if user.role == "patient":
            return redirect(
                url_for("patient.dashboard")
            )

        if user.role == "doctor":
            return redirect(
                url_for("doctor.dashboard")
            )

        if user.role == "admin":
            return redirect(
                url_for("admin.dashboard")
            )

        session.clear()

        flash(
            "Invalid account role.",
            "danger"
        )

        return redirect(url_for("auth.login"))

    return render_template(
        "auth/login.html",
        form=form
    )
```

This follows the guide's required login flow:

```text
Find user
   ↓
Verify password
   ↓
Determine role
   ↓
Create session
   ↓
Redirect to dashboard
```

The guide explicitly describes this sequence and the three possible dashboard destinations.

# 16. Step 14 — Create authentication decorators

This is one of the most important parts of the project.

Create:

```text
app/auth/decorators.py
```

```python
from functools import wraps

from flask import (
    session,
    redirect,
    url_for,
    flash
)


def login_required(view):

    @wraps(view)
    def wrapped_view(*args, **kwargs):

        if "user_id" not in session:

            flash(
                "Please log in to continue.",
                "warning"
            )

            return redirect(
                url_for("auth.login")
            )

        return view(*args, **kwargs)

    return wrapped_view


def role_required(*allowed_roles):

    def decorator(view):

        @wraps(view)
        def wrapped_view(*args, **kwargs):

            if "user_id" not in session:
                flash(
                    "Please log in to continue.",
                    "warning"
                )

                return redirect(
                    url_for("auth.login")
                )

            if session.get("role") not in allowed_roles:

                flash(
                    "You are not authorized to access this page.",
                    "danger"
                )

                return redirect(
                    url_for("main.index")
                )

            return view(*args, **kwargs)

        return wrapped_view

    return decorator
```

Now we have two concepts:

```python
@login_required
```

means:

> Are you authenticated?

And:

```python
@role_required("admin")
```

means:

> Are you authenticated **and** authorized to perform this operation?

That distinction is central to the Academy material: authentication asks **“Who are you?”**, while authorization asks **“Are you allowed to do this?”**

# 25. Step 23 — The critical security requirement

This is where many student projects fail.

**Do not rely on hiding buttons.**

For example, this is **not security**:

```jinja2
{% if session.role == "admin" %}
    <a href="/admin">Admin</a>
{% endif %}
```

Someone can simply type:

```text
/admin/dashboard
```

The Academy guide explicitly calls this out: hiding a link does not prevent someone from manually entering the URL; authorization must happen on the server. 

That's why our route contains:

```python
@role_required("admin")
```

The server checks the permission.

# 26. Step 24 — Protect individual appointments

This is even more important.

Suppose:

```text
Patient A → Appointment 15
Patient B → Appointment 16
```

Patient A must **not** be able to change:

```text
/appointments/15
```

to:

```text
/appointments/16
```

and see Patient B's information.

The guide explicitly requires checking both authentication **and ownership/authorization**. 

Example:

```python
from flask import abort, session

from app.models.appointment import Appointment


def get_patient_appointment(appointment_id):

    appointment = Appointment.query.get_or_404(
        appointment_id
    )

    current_user_id = session["user_id"]

    if appointment.patient.user_id != current_user_id:
        abort(403)

    return appointment
```

Then:

```python
@patient_bp.route("/appointments/<int:appointment_id>")
@role_required("patient")
def appointment_detail(appointment_id):

    appointment = get_patient_appointment(
        appointment_id
    )

    return render_template(
        "patient/appointment.html",
        appointment=appointment
    )
```

This is **object-level authorization**.

# 36. Step 34 — Seed an administrator

You should **not allow public registration to select `admin`**.

The registration code deliberately says:

```python
role="patient"
```

That is important.

Otherwise someone could register:

```text
email = attacker@example.com
role = admin
```

and immediately become an administrator.

Create an administrative seed script instead:

```text
seed_admin.py
```

```python
from app import create_app
from app.extensions import db
from app.models.user import User

app = create_app()

with app.app_context():

    existing = User.query.filter_by(
        email="admin@clinic.local"
    ).first()

    if existing:
        print("Admin already exists.")
    else:

        admin = User(
            email="admin@clinic.local",
            role="admin"
        )

        admin.set_password("ChangeThisImmediately123!")

        db.session.add(admin)
        db.session.commit()

        print("Administrator created.")
```

Run:

```bash
python seed_admin.py
```

Then immediately use a strong password in a real environment.
